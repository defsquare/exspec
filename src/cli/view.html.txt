function h(tag, attrs, ...children) {
  const el = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs ?? {})) {
    if (v == null || v === false) continue;
    el.setAttribute(k, v === true ? "" : v);
  }
  el.append(...children.flat(Infinity).filter((c) => c != null && c !== false));
  return el;
}
function s(tag, attrs, ...children) {
  const el = document.createElementNS("http://www.w3.org/2000/svg", tag);
  for (const [k, v] of Object.entries(attrs ?? {})) {
    if (v == null || v === false) continue;
    el.setAttribute(k, v);
  }
  el.append(...children.flat(Infinity).filter((c) => c != null && c !== false));
  return el;
}

function lines(o) {

  ]);

  return [
    ...Object.entries(o.takes).flatMap(([input, t]) => (t.rules ?? []).map((r) => ({ kind: "input", input, rule: r.rule, otherwise: r.otherwise, status: status(r.otherwise),

      say: [whenSay(e.when), thenSay(e.then)], short: typeof e.then === "string" ? "nothing changes" : "says" in e.then ? e.then.says : "changes what is kept" })),
  ];

}

const calledOf = (call) => `${call.operation} {${given(call.with)}}`;

// ---- diagrams
// What a scenario takes is drawn in blue: red is a verdict's, and a scenario that holds takes a path too.
const pathColor = "#2563eb";
// One scenario as a sequence: who calls the target, what it answers, what is read around the judged call.
function sequence(path) {
  const sc = x.scenarios.find((e) => e.path === path).scenario, ref = referenceOf(path);
  const who = (call) => call.caller ?? "nobody";
  const actors = [...new Set([...sc.arrange.map((a) => who(a.call)), who(sc.act.call)])];
  const col = 170, left = 70, targetX = left + actors.length * col, width = targetX + 430;
  const steps = [...sc.arrange.map((a, n) => ({ call: a.call, response: ref?.arrange[n]?.response })), { readings: "before" }, { call: sc.act.call, response: ref?.act.response, act: true }, { readings: "after" }];
  const readings = ref?.readings ?? [], els = [];
  let y = 64;
  for (const step of steps) {
    if (step.readings) {
      if (readings.length === 0) continue;
      const height = 22 + readings.length * 17;
      els.push(s("rect", { x: targetX + 24, y, width: 390, height, rx: 6, fill: "#f7f7f8", stroke: "#e5e7eb" }),
        s("text", { x: targetX + 36, y: y + 16, class: "small" }, `read ${step.readings} the judged call`),
        readings.map((r, n) => { const e = r[step.readings], answered = e.length === 0 ? "nothing to ask" : e.map((v) => ("unreadable" in v ? "unreadable" : v.status)).join(", ");
          return s("text", { x: targetX + 36, y: y + 33 + n * 17, class: "small" }, `${"fact" in r.of ? r.of.fact : `${r.of.shows.structure}.${r.of.shows.field.join(".")}`}  →  ${answered}`); }));
      y += height + 34;
      continue;
    }
    const ax = left + actors.indexOf(who(step.call)) * col, color = step.act ? pathColor : "#1e416e", given = JSON.stringify(step.call.with ?? {});
    els.push(s("line", { x1: ax, y1: y, x2: targetX - 6, y2: y, stroke: color, "stroke-width": step.act ? 2.5 : 1.5, "marker-end": `url(#q-${step.act ? "act" : "call"})` }),
      s("text", { x: ax + 8, y: y - 7, "font-weight": step.act ? 700 : 400 }, step.call.operation),
      given === "{}" ? null : s("text", { x: ax + 8, y: y + 15, class: "m small" }, given.length > 60 ? `${given.slice(0, 59)}…` : given),
      s("line", { x1: targetX, y1: y + 26, x2: ax + 6, y2: y + 26, stroke: "#4b5563", "stroke-dasharray": "5 4", "marker-end": "url(#q-back)" }),
      s("text", { x: targetX + 10, y: y + 30, class: "m", "font-weight": 600 }, step.response ? String(step.response.status) : "not recorded"));
    y += 62;
  }
  const lifeline = (at, label) => [s("text", { x: at, y: 24, "text-anchor": "middle", "font-weight": 600 }, label), s("line", { x1: at, y1: 34, x2: at, y2: y, stroke: "#d1d5db" })];
  const marker = (id, color) => s("marker", { id, viewBox: "0 0 10 10", refX: 9, refY: 5, markerWidth: 7, markerHeight: 7, orient: "auto-start-reverse" }, s("path", { d: "M0,0 L10,5 L0,10 z", fill: color }));
  return s("svg", { viewBox: `0 0 ${width} ${y + 10}`, width, role: "img", "aria-label": `Sequence of ${path}` },
    s("defs", null, marker("q-call", "#1e416e"), marker("q-act", pathColor), marker("q-back", "#4b5563")),
    actors.map((a, n) => lifeline(left + n * col, a)), lifeline(targetX, "the target"), els);
}

// The steps of a scenario of the roles as one sequence: arrange, before, the call (heavier), after.
// A row a step: what it says on the left, its arrow between the lifelines on the right, the status after
// the lifeline that answered. A step giving a service a condition is a mark on its provider's lifeline;
// what the target sent a provider while a step was played, a row under that step.
//   create-program        writer ───────────▶ target  201
//     ↳ POST /emails                          target ─ ─ ▶ p  200
// The lifelines are narrow so that five of them fit a panel; the diagram is never shrunk, and scrolls
// when it is wider. What does not fit a row is in its title and in the exchanges under the diagram: a row
// leads to its exchange there, where a call's operation opens its sheet.
// The steps of the path of a state the scenario starts from are one row, the state and what is seen in it;
// the row unfolds them (open: the states unfolded).
//   ▸ s                                          2 steps
//     what a reading of the system shows in s
function steps(e, open = new Set()) {
  const who = (st) => st.as ?? "nobody";
  const laneOf = (st) => providerOfService(st.service) ?? `no provider of ${st.service}`;
  const providers = [...new Set(e.steps.flatMap((st) => st.service ? [laneOf(st)] : (st.sideEffects ?? []).map((se) => se.observable)))];
  const actors = [...new Set(e.steps.filter((st) => !st.service).map(who))];
  const says = 212, gap = 56, row = 30, lanes = [...actors, "target", ...providers];
  const at = (n) => says + 24 + n * gap, target = at(actors.length), ofProvider = (name) => at(actors.length + 1 + providers.indexOf(name));
  const width = at(lanes.length - 1) + 42, els = [];
  // 7.3: the width of a character of the diagram's monospace
  const fit = (text, wide) => { const n = Math.max(4, Math.floor(wide / 7.3)); return text.length > n ? `${text.slice(0, n - 1)}…` : text; };
  const said = (text, y, more) => s("text", { x: more.x ?? 10, y: y + 4, ...more.attrs }, fit(text, (more.wide ?? says) - (more.x ?? 10) - 6));
  // over a whole row, drawn last: its title, and the way to its exchange by a click or Enter
  const leads = (y, id, whole) => {
    const over = s("rect", { x: 4, y: y - 13, width: width - 8, height: 26, rx: 4, fill: "transparent", class: "leads", tabindex: 0, role: "link", "aria-label": `${whole} — its exchange` }, s("title", null, whole));
    over.addEventListener("click", () => focusExchange(id));
    over.addEventListener("keydown", (event) => { if (event.key === "Enter") focusExchange(id); });
    return over;
  };
  let y = 56, was;
  for (const [i, st] of e.steps.entries()) {
    if (st.when !== was) {
      els.push(s("text", { x: 10, y: y + 2, class: "small", "font-weight": 600, style: "text-transform:uppercase;letter-spacing:0.08em;font-size:10.5px" }, st.when),
        s("line", { x1: 10, y1: y + 10, x2: width - 8, y2: y + 10, stroke: "#eef0f3" }));
      y += row; was = st.when;
    }
    if (st.state && st.state !== e.steps[i - 1]?.state) {
      const name = st.state, seen = states.find((one) => one.state === name)?.seen, n = e.steps.filter((one) => one.state === name).length, unfolded = open.has(name);
      const whole = `Starts from the state ${name}${seen ? ` — ${seen}` : ""}: the ${n} step${n === 1 ? "" : "s"} of its path, played first`;
      const over = s("rect", { x: 4, y: y - 13, width: width - 8, height: seen ? 44 : 26, rx: 4, fill: "transparent", class: "leads", tabindex: 0, role: "button", "aria-expanded": String(unfolded), "data-state": name, "aria-label": whole }, s("title", null, whole));
      const toggle = () => { const drawn = steps(e, new Set(unfolded ? [...open].filter((one) => one !== name) : [...open, name]));
        over.closest(".wide").replaceWith(drawn); drawn.querySelector(`[data-state="${name}"]`)?.focus(); };
      over.addEventListener("click", toggle);
      over.addEventListener("keydown", (event) => { if (event.key === "Enter") toggle(); });
      // rx 6: over the lifelines, which a state's row hides while it is folded
      els.push(s("rect", { x: 4, y: y - 13, width: width - 8, height: seen ? 44 : 26, rx: 6, fill: "#fffbeb", stroke: "#f3d9a4" }),
        said(`${unfolded ? "▾" : "▸"} ${name}`, y, { wide: width - 70, attrs: { class: "m stt", "font-weight": 700 } }),
        s("text", { x: width - 12, y: y + 4, "text-anchor": "end", class: "small" }, `${n} step${n === 1 ? "" : "s"}`),
        seen ? said(seen, y + 18, { x: 24, wide: width - 8, attrs: { class: "small" } }) : null,
        over);
      y += row + (seen ? 18 : 0);
    }
    if (st.state && !open.has(st.state)) continue;
    if (st.service) {
      const x = ofProvider(laneOf(st)), values = given(st.body), saying = `${st.service} ${st.unplayable ? `${st.is}: unplayable` : st.is}${values ? ` {${values}}` : ""}`;
      els.push(s("rect", { x: 4, y: y - 12, width: width - 8, height: 24, rx: 5, fill: "#f5f3ff" }),
        said(saying, y, { wide: x - 12, attrs: { class: "m sv", "font-weight": 600 } }),
        s("rect", { x: x - 7, y: y - 7, width: 14, height: 14, rx: 3, fill: "#7c3aed" }),
        leads(y, exchangeId(i), `${st.service} is given ${st.is}${values ? ` {${values}}` : ""}`));
      y += row;
      continue;
    }
    const from = at(actors.indexOf(who(st))), call = st.when === "call", color = call ? pathColor : "#1e416e";
    const request = st.method ? `${st.method.toUpperCase()} ${st.path}` : undefined;
    const whole = st.call ? `${calledOf(st.call)}${request ? ` — sent as ${request}` : ""}` : st.unplayable ? `unplayable: ${st.unplayable}` : request;
    const saying = said(st.call ? st.call.operation : st.unplayable ? `unplayable: ${st.unplayable}` : request, y,
      { attrs: { class: st.call ? "m sv" : "m tg", "font-weight": call ? 700 : 500 } });
    els.push(call ? s("rect", { x: 4, y: y - 13, width: width - 8, height: 26, rx: 5, fill: "#eff6ff" }) : null,
      saying,
      s("line", { x1: from, y1: y, x2: target - 6, y2: y, stroke: color, "stroke-width": call ? 2.5 : 1.5, "marker-end": `url(#r-${call ? "act" : "call"})` }),
      s("circle", { cx: from, cy: y, r: 3, fill: color }),
      s("text", { x: target + 8, y: y + 4, class: "m", "font-weight": 600 }, st.status === undefined ? (st.unplayable ? "" : "—") : String(st.status)),
      leads(y, exchangeId(i), `${whole}${st.status === undefined ? "" : ` → ${st.status}`}`));
    y += row;
    for (const [k, se] of (st.sideEffects ?? []).entries()) {
      const x = ofProvider(se.observable), sending = `${se.request.method.toUpperCase()} ${se.request.path}`;
      els.push(said(`↳ ${sending}`, y, { x: 24, attrs: { class: "m tg" } }),
        s("line", { x1: target, y1: y, x2: x - 6, y2: y, stroke: "#475569", "stroke-width": 1.5, "stroke-dasharray": "5 3", "marker-end": "url(#r-sent)" }),
        s("text", { x: x + 8, y: y + 4, class: "m", "font-weight": 600, style: se.response.status >= 400 ? "fill:#a62949" : null }, String(se.response.status)),
        leads(y, exchangeId(i, k), `sent to ${se.observable}: ${sending}, answered ${se.response.status}`));
      y += row;
    }
  }
  const service = (name) => services.find((sv) => sv.provider?.provider === name)?.service;
  const head = (name, n) => { const provider = n > actors.length, x = at(n);
    return [s("g", null, s("title", null, provider ? `${name}${service(name) ? `, which renders ${service(name)}` : ""}` : name),
        s("text", { x, y: 26, "text-anchor": "middle", "font-weight": 600, class: provider ? "tg" : null, style: "font-size:12px" }, fit(name, gap + 14))),
      s("line", { x1: x, y1: 36, x2: x, y2: y - 12, stroke: provider ? "#94a3b8" : "#d1d5db", "stroke-dasharray": provider ? "4 3" : null })]; };
  const marker = (id, color) => s("marker", { id, viewBox: "0 0 10 10", refX: 9, refY: 5, markerWidth: 7, markerHeight: 7, orient: "auto-start-reverse" }, s("path", { d: "M0,0 L10,5 L0,10 z", fill: color }));
  return h("div", { class: "wide" }, s("svg", { viewBox: `0 0 ${width} ${y - 6}`, width, height: y - 6, role: "img", "aria-label": `Steps of ${e.path}` },
    s("defs", null, marker("r-call", "#1e416e"), marker("r-act", pathColor), providers.length === 0 ? null : marker("r-sent", "#475569")),
    els.filter((el) => el?.tagName === "rect" && el.getAttribute("rx") === "5"), lanes.map(head), els.filter((el) => !(el?.tagName === "rect" && el.getAttribute("rx") === "5"))));
}

// An operation as a column of gates: each rule that can refuse is an exit to the right, the effects are
// below, a mark in the corner of each saying its verdict. With a scenario, the path it was written to take
// flows down the spine and round the rules it goes through, and a status the target answered otherwise is
// written beside the exit, in red.
function gates(o, path, picked) {
  const all = lines(o), gate = all.filter((l) => l.kind === "input" || l.kind === "requires"), effects = all.filter((l) => !gate.includes(l));
  const e = path ? scenarioAt(path) : null, sc = e ? (e.scenario ?? { proves: [e.rule] }) : null;
  // The requirements on one input share one stop, drawn where the first of them is: a mark each, by input.
  // One that does not hold keeps a gate of its own as well.
  const onInput = gate.filter((g) => g.input !== undefined);
  const stops = gate.flatMap((g) => g.input === undefined ? [{ gate: g }] : [...(g === onInput[0] ? [{ inputs: onInput }] : []), ...(bad(provenanceOf(o.operation, g.rule)) ? [{ gate: g }] : [])]);
  const proven = (g) => sc.proves.includes(g.rule);
  const held = (g) => proven(g) && !bad(provenanceOf(o.operation, g.rule)); // one that does not hold exits by its own gate
  const exitAt = sc ? stops.findIndex((st) => st.gate ? proven(st.gate) : st.inputs.some(held)) : -2; // -1: through to the end
  const answered = !path ? undefined : e?.steps ? e.steps.find((st) => st.when === "call")?.status : referenceOf(path)?.act.response.status;
  const cx = 150, bw = 260, els = [];
  let y = 20;
  const stroke = (rule) => colors[provenanceOf(o.operation, rule)] ?? colors.none;
  const mark = (rule, x, y) => s("circle", { cx: x, cy: y, r: 5, fill: stroke(rule) }, s("title", null, provenanceOf(o.operation, rule)));
  const edge = (on, service) => ({ stroke: on ? pathPaint : service ? "#7c3aed" : "#d1d5db", "stroke-width": on ? 3 : 1.5 });
  // a rule resting on a service is drawn in the specification's violet, a little taller, the service's chip at its foot
  const fill = (l, plain) => picked === l.rule ? "#d9e3f1" : l.service ? "#f5f3ff" : plain, tall = (l) => (l.service ? 64 : 44);
  const serviceChip = (left, top, name) => s("a", { href: route("service", name) }, s("title", null, `rests on ${name}`),
    s("rect", { x: left, y: top, width: name.length * 7.3 + 16, height: 16, rx: 8, fill: "#fff", stroke: "#7c3aed" }), s("text", { x: left + 8, y: top + 12, class: "m sv" }, name));
  const taken = (n) => sc && (exitAt === -1 || n <= exitAt), through = sc && exitAt === -1;
  const pathPaint = "url(#g-flow)";
  const forkColor = "#172741";
  const spine = (y1, y2, on) => s("line", { x1: cx, y1, x2: cx, y2, stroke: on ? pathPaint : "#d1d5db", "stroke-width": on ? 4 : 1.5 });
  const clip = (text, n) => (text.length > n ? `${text.slice(0, n - 1)}…` : text);
  const inputs = `${o.operation}(${Object.keys(o.takes ?? {}).join(", ")})`;
  els.push(s("rect", { x: cx - bw / 2, y, width: bw, height: 40, rx: 20, fill: "#172741" }), s("text", { x: cx, y: y + 25, "text-anchor": "middle", style: "fill:#fff", "font-weight": 600 }, inputs.length > 34 ? `${clip(o.operation, 30)}(…)` : inputs));
  y += 40;
  const exit = (y, h, out, refusal, status, answeredElse) => [
    s("line", { x1: cx + bw / 2, y1: y + h / 2, x2: cx + bw / 2 + 60, y2: y + h / 2, stroke: out ? pathPaint : "#d1d5db", "stroke-width": out ? 4 : 1.5, "marker-end": `url(#g-${out ? "on" : "off"})` }),
    s("text", { x: cx + bw / 2 + 68, y: y + h / 2 - 3, class: "m" }, refusal),
    s("text", { x: cx + bw / 2 + 68, y: y + h / 2 + 13, class: "small tg" }, status),
    answeredElse ? s("text", { x: cx + bw / 2 + 180, y: y + h / 2 + 13, style: "fill:#a62949;font-weight:600;font-size:12px" }, `answered ${answered}`) : null];
  const one = (values, word) => { const set = [...new Set(values)]; return set.length === 1 ? set[0] : `${set.length} ${word}`; };
  stops.forEach((st, n) => {
    els.push(spine(y, y + 20, taken(n)));
    y += 20;
    const out = sc && exitAt === n;
    if (st.gate) {
      const g = st.gate;
      els.push(s("a", { class: "click", href: ruleRoute(o.operation, g.rule) },
        s("rect", { x: cx - bw / 2, y, width: bw, height: tall(g), rx: 6, fill: fill(g, "#fff"), ...edge(taken(n), g.service) }), mark(g.rule, cx + bw / 2 - 14, y + 14),
        s("text", { x: cx, y: y + 19, "text-anchor": "middle", class: "m" }, clip(g.rule, 30)),
        s("text", { x: cx, y: y + 36, "text-anchor": "middle", class: "small" }, clip(g.short, 38))),
        exit(y, tall(g), out, g.otherwise, g.status === undefined ? "no status declared" : `on this target: ${g.status}`, out && answered !== undefined && answered !== g.status),
        g.service ? serviceChip(cx - bw / 2 + 12, y + 44, g.service) : null);
      y += tall(g);
      return;
    }
    // wide as an effect: a mark and a word per requirement, the word being its name without its input's
    const perRow = 4, wide = bw + 200, left = cx - bw / 2, words = st.inputs.map((g) => provenanceOf(o.operation, g.rule)), exited = out ? st.inputs.find(proven) : undefined;
    const dashed = (name) => name.replace(/([A-Z])/g, "-$1").toLowerCase(), wordOf = (g) => g.rule.startsWith(`${dashed(g.input)}-`) ? g.rule.slice(dashed(g.input).length + 1) : g.rule;
    const rows = [...new Set(st.inputs.map((g) => g.input))].flatMap((input) => {
      const marks = st.inputs.filter((g) => g.input === input);
      return Array.from({ length: Math.ceil(marks.length / perRow) }, (_, r) => ({ input: r === 0 ? input : "", marks: marks.slice(r * perRow, (r + 1) * perRow) }));
    });
    const height = 34 + rows.length * 22 + 6, status = (g) => g.status === undefined ? "no status declared" : `${g.status}`;
    els.push(s("rect", { x: left, y, width: wide, height, rx: 6, fill: "#fff", ...edge(taken(n)) }),
      s("text", { x: left + 12, y: y + 21, class: "m" }, "inputs"),
      s("text", { x: left + 70, y: y + 21, class: "small" }, `${words.filter(good).length} of ${words.length} hold`),
      s("text", { x: left + wide - 12, y: y + 21, "text-anchor": "end", class: "small" }, `otherwise ${one(st.inputs.map((g) => g.otherwise), "refusals")} · on this target: ${one(st.inputs.map(status), "statuses")}`),
      rows.map((row, r) => [
        s("text", { x: left + 12, y: y + 34 + r * 22 + 14, class: "small", "font-weight": 600 }, clip(row.input, 13)),
        row.marks.map((g, m) => s("a", { class: "click", href: ruleRoute(o.operation, g.rule) },
          s("title", null, `${g.rule} — ${g.short}`),
          picked === g.rule || g === exited ? s("rect", { x: left + 100 + m * 88, y: y + 34 + r * 22, width: 86, height: 20, rx: 10, fill: "#d9e3f1" }) : null,
          s("circle", { cx: left + 110 + m * 88, cy: y + 34 + r * 22 + 10, r: 5, fill: stroke(g.rule) }),
          s("text", { x: left + 120 + m * 88, y: y + 34 + r * 22 + 14, class: "small" }, clip(wordOf(g), 11))))]),
      exited ? [s("line", { x1: left + wide, y1: y + height / 2, x2: left + wide + 40, y2: y + height / 2, stroke: pathPaint, "stroke-width": 4, "marker-end": "url(#g-on)" }),
        s("text", { x: left + wide + 46, y: y + height / 2 + 4, class: "small", style: answered !== undefined && answered !== exited.status ? "fill:#a62949;font-weight:600" : "" }, answered !== undefined && answered !== exited.status ? `answered ${answered}` : status(exited))] : null);
    y += height;
  });
  // An effect as a box under the spine, or in an outcome's column: a rule the flow forks on carries the fork's mark.
  const effectBox = (e, left, top, width, on, forks) => s("a", { class: "click", href: ruleRoute(o.operation, e.rule) }, s("title", null, `${e.rule} — ${e.short}`),
    s("rect", { x: left, y: top, width, height: tall(e), rx: 6, fill: fill(e, "#f7f7f8"), ...edge(on, e.service) }), mark(e.rule, left + width - 14, top + 14),
    forks ? s("text", { x: left + width - 26, y: top + 18, "text-anchor": "end", style: `fill:${forkColor};font-weight:700;font-size:15px` }, "⑂") : null,
    s("text", { x: left + 12, y: top + 19, class: "m" }, clip(e.rule, Math.floor((width - (forks ? 52 : 36)) / 7.3))),
    s("text", { x: left + 12, y: top + 36, class: "small" }, clip(e.short, Math.floor((width - 24) / 6.6))),
    e.service ? serviceChip(left + 12, top + 44, e.service) : null);
  // with the roles' reading, the spine carries the effects on every accepted call and what the operation answers;
  // the effects with a `when` are drawn on the outcomes the recordings were seen to reach, or, none seen yet, by their input
  const br = o.branches, outcomes = br?.outcomes ?? [], slug = path?.split("/")[1];
  const onSpine = br ? effects.filter((l) => l.kind !== "effect" || o.flows.line.includes(l.rule)) : effects;
  const byRule = (rule) => effects.find((l) => l.rule === rule);
  onSpine.forEach((e) => {
    els.push(spine(y, y + 20, through));
    y += 20;
    els.push(effectBox(e, cx - bw / 2, y, bw + 200, through, false));
    y += tall(e);
  });
  const t = triggerOf(o.operation), accepted = acceptedOf(t), answers = typeof o.answers === "string" ? o.answers : o.answers?.structure;
  // an exit's look is its status's: two outcomes ending in one status end alike
  const exitColors = ["#172741", "#0f766e", "#b45309", "#be185d"], exitColor = (status) => exitColors[Math.max(0, (accepted ?? []).indexOf(status)) % exitColors.length];
  let width = 640;
  if (outcomes.length > 0) {
    const cw = 250, gap = 16, left = (n) => cx - bw / 2 + n * (cw + gap), mid = (n) => left(n) + cw / 2;
    const taking = outcomes.findIndex((p) => through && p.scenarios.includes(slug));
    width = Math.max(width, left(outcomes.length) + 20);
    els.push(spine(y, y + 16, through && taking !== -1));
    y += 16;
    // the spine forks: a bar across the columns, a drop to each
    els.push(s("line", { x1: Math.min(cx, mid(0)), y1: y, x2: mid(outcomes.length - 1), y2: y, stroke: "#d1d5db", "stroke-width": 1.5 }),
      taking === -1 ? null : s("line", { x1: Math.min(cx, mid(taking)), y1: y, x2: Math.max(cx, mid(taking)), y2: y, stroke: pathPaint, "stroke-width": 4 }));
    if (through && taking === -1 && answered !== undefined) els.push(s("text", { x: cx + 12, y: y - 4, style: "fill:#a62949;font-weight:600;font-size:12px" }, accepted?.includes(answered) ? "on no outcome seen" : `answered ${answered}`));
    outcomes.forEach((p, n) => els.push(s("line", { x1: mid(n), y1: y, x2: mid(n), y2: y + 54, stroke: n === taking ? pathPaint : "#d1d5db", "stroke-width": n === taking ? 4 : 1.5 }),
      s("text", { x: mid(n) + 10, y: y + 26, class: "small", "font-weight": 600 }, `outcome ${n + 1} · ${plural(p.scenarios.length, "scenario")}`, s("title", null, p.scenarios.join(", "))),
      p.unsure.length === 0 ? null : s("text", { x: mid(n) + 10, y: y + 42, class: "small" }, clip(`unsure: ${p.unsure.join(", ")}`, 20), s("title", null, `unsure: ${p.unsure.join(", ")}`))));
    y += 54;
    // row by row across the columns, so that the arrows go along a row as they do among the gates
    const met = outcomes.map((p) => p.met.flatMap((rule) => byRule(rule) ?? []));
    const rows = Math.max(...met.map((m) => m.length));
    const ends = outcomes.map(() => y);
    for (let r = 0; r < rows; r++) {
      const height = Math.max(...met.map((m) => (m[r] ? tall(m[r]) : 0)));
      met.forEach((m, n) => {
        if (!m[r]) return;
        els.push(s("line", { x1: mid(n), y1: y, x2: mid(n), y2: y + 14, stroke: n === taking ? pathPaint : "#d1d5db", "stroke-width": n === taking ? 4 : 1.5 }),
          effectBox(m[r], left(n), y + 14, cw, n === taking, br.forks.includes(m[r].rule)));
        ends[n] = y + 14 + tall(m[r]);
      });
      y += 14 + height;
    }
    // each column ends in its exit, the exits on one line
    outcomes.forEach((p, n) => {
      const on = n === taking, color = exitColor(p.status);
      els.push(s("line", { x1: mid(n), y1: ends[n], x2: mid(n), y2: y + 20, stroke: on ? pathPaint : "#d1d5db", "stroke-width": on ? 4 : 1.5 }),
        s("rect", { x: left(n), y: y + 20, width: cw, height: 40, rx: 20, fill: "#fff", stroke: on ? pathPaint : color, "stroke-width": on ? 4 : 2.5 }),
        s("text", { x: mid(n), y: y + 45, "text-anchor": "middle" }, "accepted · ", s("tspan", { style: `fill:${color};font-weight:600` }, String(p.status))),
        answers ? s("text", { x: mid(n), y: y + 76, "text-anchor": "middle", class: "small" }, `answers ${answers}`) : null);
    });
    y += answers ? 82 : 66;
    const neverMet = br.neverMet, unanswered = br.accepted.filter((ex) => !ex.seen);
    if (neverMet.length > 0) {
      els.push(s("text", { x: cx - bw / 2, y: y + 18, class: "small" }, "never met in a recording: ", commas(neverMet.map((rule) => s("a", { href: ruleRoute(o.operation, rule) }, s("tspan", { class: "m", style: `fill:${forkColor}` }, rule))))));
      y += 22;
    }
    unanswered.forEach((ex) => {
      els.push(s("text", { x: cx - bw / 2, y: y + 18, class: "small" }, s("tspan", { style: `fill:${exitColor(ex.status)};font-weight:600` }, String(ex.status)), " — no recording"));
      y += 22;
    });
  } else {
    const flows = o.flows, groups = flows ? [...flows.variants.map((v) => ({ by: `by ${v.on}`, cases: v.cases })), ...(flows.turns.length > 0 ? [{ by: "by the situation", cases: flows.turns }] : [])] : [];
    if (groups.length > 0) {
      els.push(spine(y, y + 20, through));
      y += 20;
      els.push(s("text", { x: cx - bw / 2, y: y + 14, class: "small", "font-weight": 600 }, "not seen yet: no accepted recording judged on its conditions"));
      y += 22;
      groups.forEach((g) => {
        els.push(s("text", { x: cx - bw / 2 + 12, y: y + 16, class: "m small" }, g.by));
        y += 22;
        g.cases.forEach((c) => {
          els.push(s("text", { x: cx - bw / 2 + 24, y: y + 14, class: "small" }, clip(`when ${c.when}`, 70)));
          y += 20;
          c.effects.forEach((rule) => {
            const e = byRule(rule);
            if (!e) return;
            els.push(effectBox(e, cx - bw / 2 + 24, y, bw + 176, false, false));
            y += tall(e) + 8;
          });
        });
      });
    }
    els.push(spine(y, y + 20, through));
    y += 20;
    els.push(s("rect", { x: cx - bw / 2, y, width: bw, height: 40, rx: 20, fill: "#fff", stroke: through ? pathPaint : "#172741", "stroke-width": through ? 4 : 1.5 }),
      s("text", { x: cx, y: y + 25, "text-anchor": "middle" }, "accepted", t && accepted.length > 0 ? s("tspan", { class: "tg" }, ` · ${accepted.join(", ")}`) : "", answers ? ` · answers ${answers}` : ""),
      through && answered !== undefined && accepted !== undefined && !accepted.includes(answered) ? s("text", { x: cx + bw / 2 + 12, y: y + 25, style: "fill:#a62949;font-weight:600;font-size:12px" }, `answered ${answered}`) : null);
  }
  const marker = (id, color, size) => s("marker", { id, viewBox: "0 0 10 10", refX: 9, refY: 5, markerUnits: "userSpaceOnUse", markerWidth: size, markerHeight: size, orient: "auto" }, s("path", { d: "M0,0 L10,5 L0,10 z", fill: color }));
  return s("svg", { viewBox: `0 0 ${width} ${y + 60}`, width, role: "group", "aria-label": `Gates of ${o.operation}` }, s("defs", null, marker("g-on", pathColor, 16), marker("g-off", "#d1d5db", 10),
    // a band from blue to violet to cyan, repeated down the page and sliding: a line is too thin to carry a gradient of its own
    s("linearGradient", { id: "g-flow", gradientUnits: "userSpaceOnUse", x1: 0, y1: 0, x2: 0, y2: 120, spreadMethod: "repeat" },
      s("stop", { offset: 0, "stop-color": pathColor }), s("stop", { offset: 0.4, "stop-color": "#a855f7" }), s("stop", { offset: 0.7, "stop-color": "#22d3ee" }), s("stop", { offset: 1, "stop-color": pathColor }),
      matchMedia("(prefers-reduced-motion: reduce)").matches ? null : s("animateTransform", { attributeName: "gradientTransform", type: "translate", from: "0 0", to: "0 120", dur: "1.6s", repeatCount: "indefinite" }))), els);
}

// ---- the sheets
function structureSheet(name) {
  const st = x.structures.find((e) => e.structure === name);
  const by = (verb) => data.placed.filter((p) => p.touches.some((t) => t.structure === name && t.as.some((a) => a.verb === verb))).map((p) => p.operation);
  const referring = x.structures.filter((e) => Object.values(e.fields).some((f) => f.structure === name)).map((e) => e.structure);
  const shown = x.projections.filter((p) => p.shows.some((sh) => sh.structure === name));
  return [
    h("p", { class: "eyebrow" }, "Structure"), h("h1", null, name),
    h("p", null, "Identified by ", code(st.identity), ".", referring.length ? [" Referred to by ", commas(referring.map(toStructure)), "."] : ""),
    h("h2", null, "Fields"),
    table(["Field", "Is", "Constraints"], Object.entries(st.fields).map(([n, f]) => [code(n),
      f.structure ? [f.many ? "many " : "one ", toStructure(f.structure)] : f.type,
      f.structure ? "" : Object.entries(f).filter(([k]) => k !== "type").map(([k, v]) => `${k}: ${v}`).join(", ")])),
    h("h2", null, "Its life"),
    table(["", "Operations"], [["answers", "Answered by"], ["changes", "Changed by"], ["takes", "Named by"], ["requires", "Guards"]].map(([verb, label]) => [label,
      by(verb).length ? h("div", { class: "chips" }, by(verb).map((o) => h("a", { class: "chip", href: route("operation", o) }, o))) : h("span", { class: "muted" }, "nothing")])),
    h("h2", null, "On the wire"),
    table(["Projection", "Interface", "At"], shown.map((p) => { const i = x.interfaces.find((e) => e.interface === p.interface);
      return [code(p.projection), i ? code(`${i.method.toUpperCase()} ${i.pattern}`) : code(p.interface), code(p.shows.find((sh) => sh.structure === name).at.join("."))]; })),
  ];
}

// the panel of one rule: what it says, the fact it rests on, and each scenario that judges it
function rulePanel(o, line) {
  if (roles) return roleRulePanel(o, line);
  const j = judged(o.operation, line.rule), f = line.fact ? x.facts.find((e) => e.fact === line.fact) : null;
  return [
    h("a", { class: "close", href: route("operation", o.operation) }, "Close", h("kbd", null, "esc")),
    h("p", { class: "eyebrow" }, `Rule of ${o.operation}`), h("h2", null, code(line.rule)), h("p", null, line.say),
    h("p", null, provenanceDot(j?.provenance ?? "declared-only"), h("b", null, j?.provenance ?? "declared-only"), j?.reason ? ` — ${j.reason}` : ""),
    line.otherwise ? h("p", { class: "muted" }, code(line.otherwise), line.status === undefined ? " has no status declared." : [" is answered ", code(line.status), " on the wire."]) : null,
    f ? [h("h3", null, "The fact"), h("p", null, f.condition.length === 0 ? "Declared without a condition: nothing can read it." : commas(f.condition.map((t) => "caller" in t ? ["the caller is ", code(t.caller)]
      : [toStructure(t.structure), " where ", Object.entries(t.where).map(([k, v]) => `${k} is the call's ${v[1]}`).join(", "), t.field ? `, ${t.field.join(".")}` : "", " must be ", code(t["must-be"].join(" "))])))] : null,
    h("h3", null, "The scenarios", h("span", { class: "keys" }, keys(["↑↓", "draw its path"], ["⏎", "open"]))),
    (j?.by ?? []).length === 0 ? h("p", { class: "muted" }, "None judges it.") : j.by.map((b) => scenarioCard(b.scenario,
      h("p", null, scenarioName(b.scenario), provenanceDot(b.verdict), b.verdict, b.reason ? `: ${b.reason}` : ""),
      sequence(b.scenario))),
  ];
}

}

// the panel of one scenario: what it proves, what the references say of it, and what happened

  const call = (written, played, label) => h("div", { class: "card" },
    h("p", null, h("span", { class: "word" }, `${label}  `), toOperation(written.call.operation), written.call.caller ? ` as ${written.call.caller}` : " as nobody", played ? [" — answered ", code(played.response.status)] : ""),
    Object.keys(written.call.with ?? {}).length ? json(written.call.with) : null,
    played ? h("details", { class: "body" }, h("summary", null, "What the target answered"), json(played.response.body)) : null);
  return [
    h("a", { class: "close", href: back }, "Close", h("kbd", null, "esc")),
    h("p", { class: "eyebrow" }, `Scenario of ${name}`), h("h2", null, sc.scenario), h("p", null, sc.says),
    h("p", null, "Proves ", commas(sc.proves.map((rule) => h("a", { href: ruleRoute(name, rule) }, code(rule)))), "."),
    ref ? [h("h3", null, "Verdicts"), table(["Claim", "Verdict", "Reason"], verdictsOf(path).map((v) => [code(v.claim), [provenanceDot(v.verdict), v.verdict], v.reason ?? ""])),
      h("p", { class: "muted" }, `Recorded on ${ref.run.target} at ${ref.run.version}, ${ref.run.at}. ${ref.noise.length} place(s) differ between two executions.`)]
      : h("p", { class: "muted" }, "No reference yet: nothing is judged."),
    h("h3", null, "What happens"), sequence(path),
    h("h3", null, "The calls", h("span", { class: "keys" }, keys(["↑↓", "body"], ["←→", "fold"]))), sc.arrange.map((a, n) => call(a, ref?.arrange[n], "ARRANGE")), call(sc.act, ref?.act, "ACT"),
  ];
}

  const j = e.judgement, line = (st) => st.unplayable ? `${st.when} as ${st.as ?? "nobody"}: unplayable, ${st.unplayable}`
    : !st.method ? `${st.when}  as ${st.as ?? "nobody"}: not played`
    : `${st.when}  ${st.method.toUpperCase()} ${st.path}${st.query ? `?${new URLSearchParams(Object.entries(st.query).map(([k, v]) => [k, typeof v === "string" ? v : JSON.stringify(v)]))}` : ""}  as ${st.as ?? "nobody"}${st.status === undefined ? "" : `  → ${st.status}`}`;
  return [
    h("a", { class: "close", href: back }, "Close", h("kbd", null, "esc")),
    h("p", { class: "eyebrow" }, `Scenario of ${e.operation}`), h("h2", null, e.path.split("/")[1]),
    h("p", null, "Exercises ", h("a", { href: ruleRoute(e.operation, e.rule) }, code(e.rule)), "."),
    e.startsFrom ? h("p", null, "Starts from the state ", commas(e.startsFrom.map((name) => [toState(name), states.find((one) => one.state === name)?.seen ? ` (${states.find((one) => one.state === name).seen})` : ""])), h("span", { class: "muted" }, " — its path's steps are played first, as arrange steps.")) : null,
    e.situation ? h("p", null, e.situation, h("span", { class: "muted" }, ` — reached: ${e.reached}`)) : null,
    h("h3", null, "The judgement"),
    j ? [h("p", null, judgementOf(j)), h("p", null, j.says)] : h("p", { class: "muted" }, e.steps.some((st) => st.status !== undefined) ? "Not judged yet." : "Not played yet: nothing is judged."),
    h("h3", null, "What happens"), steps(e),
    h("h3", null, "The exchanges", h("span", { class: "keys" }, keys(["↑↓", "body"], ["←→", "fold"]))), e.steps.map((st, i) => st.service ? givenCard(st, exchangeId(i)) : [h("div", { class: "card", id: exchangeId(i), tabindex: -1 }, st.call ? callLine(st) : null, h("p", st.call ? { style: "color:var(--target)" } : null, code(line(st))),
      st.body === undefined ? null : h("details", { class: "body" }, h("summary", null, "What was sent"), json(st.body)),
      st.answer === undefined || st.answer === null ? null : h("details", { class: "body" }, h("summary", null, "What the target answered"), json(st.answer))),

// the card of a step's exchange, and of the k-th call it made the target send: what a row of the diagram leads to
const exchangeId = (i, k) => (k === undefined ? `exchange-${i}` : `exchange-${i}-${k}`);
function focusExchange(id) {
  const card = document.getElementById(id);
  if (!card) return;
  card.scrollIntoView({ block: "center", behavior: matchMedia("(prefers-reduced-motion: reduce)").matches ? "auto" : "smooth" });
  card.focus({ preventScroll: true });
}
// a step calling an operation, in the specification's words: the operation opens its sheet
function callLine(st) {
  const { operation } = st.call;
  return h("p", null, h("span", { class: "word" }, "CALLS "),
    st.interface ? h("a", { href: route("operation", st.interface), style: "font-weight:600" }, operation) : h("b", null, operation),
    " ", code(`{${given(st.call.with)}}`));
}
// a step giving a service a condition, in the specification's words
function givenCard(st, id) {
  const says = serviceAt(st.service)?.conditions.find((c) => c.condition === st.is)?.says;
  return h("div", { class: "card given", id, tabindex: -1 },
    h("p", null, h("span", { class: "word" }, "GIVEN "), h("a", { href: route("service", st.service), style: "color:var(--service);font-weight:600" }, st.service), " ", h("span", { class: "cond" }, st.is),
      st.body === undefined ? "" : [" ", code(given(st.body))], st.unplayable ? h("span", { class: "muted" }, ` — unplayable: ${st.unplayable}`) : ""),
    says ? h("p", { class: "muted" }, says) : null);
}
// what the target sent a provider while a step was played, under that step: a digest stays a digest
function sentCard({ observable, request: { method, path, ...sent }, response }, id) {
  return h("div", { class: "card sent", id, tabindex: -1 },
    h("p", null, h("span", { class: "word" }, "SENT TO "), h("b", { style: "color:var(--target)" }, observable), " ", code(`${method.toUpperCase()} ${path}`), " — answered ",
      h("code", { style: response.status >= 400 ? "color:#a62949" : null }, String(response.status))),
    Object.keys(sent).length === 0 ? null : h("details", { class: "body" }, h("summary", null, "What was sent"), json(sent)),
    response.body === undefined || response.body === null ? null : h("details", { class: "body" }, h("summary", null, "What was answered"), json(response.body)));
}
// a service of the specification: its conditions, the rules resting on it, its doubts; set apart, the provider rendering it here
function serviceSheet(name) {
  const sv = serviceAt(name);
  if (!sv) return [h("p", { class: "eyebrow" }, "Service"), h("h1", null, name), h("p", { class: "muted" }, "No service of this name parses.")];
  const p = sv.provider;

function overview() {
  const bar = (title, counts) => { const total = Object.values(counts).reduce((a, b) => a + b, 0);
    return [h("p", { class: "eyebrow" }, title), h("div", { class: "bar" }, Object.entries(counts).map(([word, n]) => h("i", { style: `background:${colors[word] ?? colors.none};width:${total ? (100 * n) / total : 0}%` }))),
      h("p", null, Object.entries(counts).map(([word, n]) => [provenanceDot(word), `${n} ${word}   `]))]; };
  const why = (r) => r.reason ?? [...r.by.filter((b) => b.verdict === r.provenance), ...r.by].map((b) => b.reason ?? b.says).filter(Boolean)[0] ?? "";
  const notHeld = data.rules.filter((r) => !good(r.provenance));
  // what is not held without being wrong: a shared rule is one line per word, its operations beside it; a rule of
  // one operation is a line of its own, two operations naming a rule alike not making it one rule
  //   shared r:  [{operation: "a", rule: "r"}, {operation: "b", rule: "r"}]  ->  r | a, b
  const shared = new Set((data.common?.rules ?? []).map((r) => r.rule));
  const notShownByRule = [...notHeld.filter((r) => !bad(r.provenance)).reduce((byRule, r) => {
    const key = shared.has(r.rule) ? `${r.rule} ${r.provenance}` : `${r.operation} ${r.rule}`;
    return byRule.set(key, [...(byRule.get(key) ?? []), r]);
  }, new Map()).values()];
  const whys = (rs) => { const all = [...new Set(rs.map(why))].filter(Boolean);
    return all.length < 2 ? all : [all[0], h("span", { class: "muted" }, ` · ${all.length - 1} other ${all.length === 2 ? "reason" : "reasons"}, under each operation`)]; };
  return [
    h("h1", null, "The specification of ", h("b", null, x.target?.target ?? data.name), "."),
    h("p", { class: "muted" }, roles ? ["Walk it from a structure to its operations, then to the scenarios that exercise them. Each rule carries the judge's word on its evidence. ", h("a", { href: "#/common" }, "What holds for every operation"), "."]
      : "Walk it from a structure to its operations, then to the scenarios that prove them. Every verdict is computed from the references, each time this page is made."),
    services.length === 0 ? null : [h("h2", null, "The system and its services"), system(), h("p", { class: "muted" }, "Its actors are the names the scenarios call as; a call made as nobody is not drawn."), operationsTable()],
    h("h2", null, "What holds"), bar("Rules", data.counts.rules), data.counts.bindings ? bar("Bindings", data.counts.bindings) : null,
    h("h2", null, roles ? "Rules that do not hold" : "Rules refuted"),
    table(["Operation", "Rule", "Provenance", "Why"], notHeld.filter((r) => bad(r.provenance)).map((r) => [h("span", { class: "unbroken" }, toOperation(r.operation)), h("a", { class: "unbroken", href: ruleRoute(r.operation, r.rule) }, code(r.rule)), h("span", { class: "unbroken" }, provenanceDot(r.provenance), r.provenance), why(r)])),
    h("h2", null, roles ? "Rules not shown" : "Rules not verified"),
    table(["Rule", "Provenance", "Operations", "Why"], notShownByRule.map((rs) => [h("span", { class: "unbroken" }, code(rs[0].rule)), h("span", { class: "unbroken" }, provenanceDot(rs[0].provenance), rs[0].provenance), commas(rs.map((r) => h("a", { href: ruleRoute(r.operation, r.rule) }, r.operation))), whys(rs)])),
  ];
}

// ---- the keyboard: & goes to the list; in it the arrows move and fold, Enter opens; among the gates the arrows
// move from row to row and along a row, among the scenarios of a rule from one to the next, in a scenario from
// Close to each body sent or answered, which they fold; Escape closes the panel on top
let byKeyboard = false;
window.addEventListener("pointerdown", () => { byKeyboard = false; }, true);
const rowsShown = () => [...document.querySelectorAll("nav .row")].filter((r) => !r.matches("a") || r.parentElement.open);
const focusOf = (row) => (row?.matches("div") ? row.querySelector("a") : row);
window.addEventListener("keydown", (e) => {
  const row = document.activeElement?.closest("nav .row"), gate = document.activeElement?.closest("svg a.click");
  const stopsOf = ".panel .close, .panel a.card, .panel details.body > summary", stop = document.activeElement?.closest(stopsOf);
  byKeyboard = true;
  const step = { ArrowDown: 1, ArrowUp: -1 }[e.key], side = { ArrowRight: 1, ArrowLeft: -1 }[e.key];
  if (e.key === "Escape") document.querySelector(".close")?.click();
  else if (e.key === "&" || e.key === "1") { e.preventDefault(); focusOf(rowsShown()[0])?.focus(); }
  else if (row) {
    const rows = rowsShown(), fold = row.matches("summary") ? row.parentElement : null;
    if (step) { e.preventDefault(); focusOf(rows[rows.indexOf(row) + step])?.focus(); }
    else if (side && fold) { e.preventDefault(); fold.open = side === 1; }
    else if (side === -1 && row.matches("a")) { e.preventDefault(); row.parentElement.querySelector("summary").focus(); }
    else if (e.key === "Enter" && fold) { e.preventDefault(); location.hash = row.querySelector("a").getAttribute("href"); }
  } else if (stop && (step || (side && stop.matches("summary")))) {
    e.preventDefault();
    const stops = [...document.querySelectorAll(stopsOf)];
    if (step) stops[stops.indexOf(stop) + step]?.focus(); else stop.parentElement.open = side === 1;
  } else if (gate && (step || side)) {
    e.preventDefault();
    // the marks of one input are a row: a mark picked is a little taller, so two tops within 12px are one row
    const gates = [...gate.ownerSVGElement.querySelectorAll("a.click")], n = gates.indexOf(gate), top = (g) => g.getBoundingClientRect().top;
    const beside = (a, b) => Math.abs(top(a) - top(b)) < 12, above = gates.slice(0, n).reverse().find((g) => !beside(g, gate));
    const to = side ? (gates[n + side] && beside(gates[n + side], gate) ? gates[n + side] : null)
      : step === 1 ? gates.slice(n + 1).find((g) => !beside(g, gate)) : above && gates.find((g) => beside(g, above));
    to?.focus();
  }
});
