// h("a", { href: "#" }, "text", node) -> an element; null and false children are left out
function h(tag, attrs, ...children) {
  const el = document.createElement(tag);
  for (const [k, v] of Object.entries(attrs ?? {})) if (v != null && v !== false) el.setAttribute(k, v);
  el.append(...children.flat(Infinity).filter((c) => c != null && c !== false));
  return el;
}

function citedBy(concept, name) {

    word,

  ])));
}

// the side column: the names of one concept; a scenario's under its operation
function side(concept, current) {
  const { all, name } = concepts[concept];
  const filter = h("input", { type: "search", placeholder: "Filter", "aria-label": "Filter" });
  const list = h("div");
  const item = (n, text) => h("li", null, h("a", { href: route(concept, n), "aria-current": n === current ? "page" : null }, text ?? n));
  const fill = () => {
    const names = all.map(name).filter((n) => n.toLowerCase().includes(filter.value.toLowerCase())).sort();
    const groups = concept !== "scenario" ? [[null, names]] : [...new Set(names.map((n) => n.split("/")[0]))].map((g) => [g, names.filter((n) => n.split("/")[0] === g)]);
    list.replaceChildren(...groups.flatMap(([group, of]) => [group ? h("h4", null, group) : null, h("ul", null, of.map((n) => item(n, group ? n.slice(group.length + 1) : n)))]).filter(Boolean));
  };
  filter.addEventListener("input", fill);
  fill();
  return h("aside", null, filter, list);
}

  const page = document.getElementById("page");
  document.getElementById("nav").replaceChildren(
    ...Object.entries(concepts).map(([c, { title, all }]) => h("a", { href: route(c), "aria-current": c === concept ? "page" : null }, title, " ", h("small", null, String(all.length)))),
    h("a", { href: "#/findings", "aria-current": concept === "findings" ? "page" : null }, "Problems and gaps ", h("small", null, String(data.problems.length + data.gaps.length))));
  if (Object.hasOwn(concepts, concept)) {
    const first = concepts[concept].all[0];
    const shown = name === undefined ? first : find(concept, name);
    const shownName = shown === undefined ? undefined : concepts[concept].name(shown);
    page.className = "page";
    page.replaceChildren(side(concept, shownName), h("main", null, shown === undefined ? h("p", { class: "muted" }, name === undefined ? "None declared." : `No ${concept} named ${name}.`) : views[concept](shown)));
    page.querySelector("aside a[aria-current]")?.scrollIntoView({ block: "nearest" });
  } else {
    page.className = "page wide";
    page.replaceChildren(h("main", null, concept === "findings" ? findings() : overview()));
  }
  document.title = `specimen — ${name ?? x.target?.target ?? data.name}`;
  window.scrollTo(0, 0);
}
document.getElementById("footer").textContent = `specimen — ${data.dir}. This page holds what the target answered, tokens included: mind who it is sent to.`;
window.addEventListener("hashchange", render);
render();
